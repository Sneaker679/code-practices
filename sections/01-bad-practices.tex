\begin{frame}{Toyota unintended acceleration (2013)}
    \source{srs2013toyota}

    \begin{columns}[T]
        \column{0.5\textwidth}
        \begin{itemize}
            \item Fatal crash blamed on sudden acceleration
            \item Experts reviewed the engine software
            \item ``Spaghetti code'': too complex to test
            \item \textbf{Jury found Toyota liable}
        \end{itemize}

        \column{0.46\textwidth}
        \centering
        % Every function reads and writes the same global state
        \begin{tikzpicture}[scale=0.9,
            fn/.style={draw=talkdim, rounded corners=1pt, fill=codebg, text=talkfg,
                       font=\tiny\ttfamily, inner sep=2pt},
            link/.style={<->, talkbad!70!talkbg, line width=0.5pt}]
            \node[circle, draw=talkbad, thick, fill=talkbad!15!talkbg, text=talkbad,
                  font=\scriptsize\bfseries, align=center, minimum size=1.2cm] (g)
                  {global\\state};
            \foreach \a/\n in {90/brake, 135/fuel, 180/idle, 225/cruise,
                               270/sensor, 315/logger, 0/timer, 45/throttle}
                \node[fn] (\n) at (\a:1.6) {\n()};
            \foreach \n in {brake, fuel, idle, cruise, sensor, logger, timer, throttle}
                \draw[link] (\n) -- (g);
            \draw[talkdim] (brake) to[bend left=25] (timer);
            \draw[talkdim] (idle) to[bend right=25] (sensor);
            \draw[talkdim] (fuel) to[bend left=40] (logger);
        \end{tikzpicture}

        \pause
        \vspace{0.5em}
        \bigstat{$\sim$10\,000}{global variables}
    \end{columns}

    \pause
    \vfill

    \takeaway{When everything can change everything, nothing can be tested.}

\end{frame}
