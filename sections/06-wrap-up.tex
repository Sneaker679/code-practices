\section[Wrap-up]{Wrap-up}

\begin{frame}{When not to apply this}

    \begin{itemize}
        \item Exploratory work you will throw away
        \vfill
        \item Code that has never changed or been copied: wait for the second or third time
            (\textit{rule of three})
        \vfill
        \item Too much abstraction is also bad: \textit{YAGNI}, ``You Aren't Gonna Need It''
    \end{itemize}

    \takeaway{Structure should pay for itself, especially on the long term.}
\end{frame}




\begin{frame}{Summary}

    \takeaway{Good design = low coupling + high cohesion.}

    \begin{itemize}
        \item Separate logic from input/output
        \item Put behaviour with its data
        \item Know as little as possible about others
        \item One algorithm, swappable steps
        \item Don't create what you depend on
        \item Keep contracts honest
    \end{itemize}

    \vfill
\end{frame}
