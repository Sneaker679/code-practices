\begin{frame}{Two goals: low coupling, high cohesion}
    \source{larman1997applying,martin2000principles}

    \begin{columns}[T]
        \column{0.48\textwidth}
        \textbf{High cohesion}
        \begin{itemize}
            \item What is \textit{inside} a class belongs together
            \item One clear job
            \item Less to keep in mind (\textit{cognitive load})
        \end{itemize}

        \column{0.48\textwidth}
        \textbf{Low coupling}
        \begin{itemize}
            \item Classes know little about \textit{each other}
            \item Few, simple dependencies
            \item Fewer edits per change (\textit{change amplification})
        \end{itemize}
    \end{columns}

    \takeaway{Every principle that follows serves one of these two goals.}
    \begin{center}
    \small They come from two families: GRASP and SOLID. We keep the essentials.
    \end{center}
\end{frame}

    \takeaway{Keep calculations free of reading, writing and printing.}
    \begin{center}
    Logic you can test without a file; files you can change without touching the logic.

\begin{frame}[fragile]{One algorithm, swappable steps}

\begin{frame}[fragile]{Don't create what you depend on}
    \framesubtitle{Dependency Inversion \textperiodcentered{} Protected Variations}

    \begin{columns}[T]
        \column{0.48\textwidth}
        \beforelabel
        \inputminted[highlightlines={3}]{python}{code/design/inject_before.py}

        \pause
        \column{0.48\textwidth}
        \afterlabel
        \inputminted[highlightlines={2,3,9,10,11}]{python}{code/design/inject_after.py}
    \end{columns}

    \vfill

    \takeaway{Receive your tools, don't build them, and swap in a fake for tests.}
    \begin{center}
    Strategy decides \textit{what} varies; injection decides \textit{who} creates it.
    \end{center}
\end{frame}
