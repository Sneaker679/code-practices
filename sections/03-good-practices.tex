        \begin{itemize}
            \setlength{\itemsep}{1.1em}
            \item<1-> {\large\textbf{Keep it simple (KISS)}}\\
                {\small\textcolor{talkdim}{Clear beats clever: a smart one-liner costs the next reader ten minutes}}
            \item<2-> {\large\textbf{You aren't gonna need it (YAGNI)}}\\
                {\small\textcolor{talkdim}{No options or classes ``just in case''}}
            \item<3-> {\large\textbf{Unit tests}}\\
                {\small\textcolor{talkdim}{A few tests let you change code without fear}}
            \item<4-> {\large\textbf{Make every run reproducible}}\\
                {\small\textcolor{talkdim}{Fixed seeds (}\mintinline{python}{np.random.default_rng(42)}\textcolor{talkdim}{), git, recorded library versions}}
        \end{itemize}
    \end{minipage}
